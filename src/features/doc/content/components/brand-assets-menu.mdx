## Features

- Copy mark and logotype SVGs to the clipboard with toast feedback.
- Link out to brand guidelines and asset downloads from one menu.
- Wrap any element to use it as a right-click or press-and-hold trigger.

## Usage

```tsx
// Using @heyompatil/ui package
import { BrandAssetsMenu } from "@heyompatil/ui"

// Or using shadcn CLI / manual import
import { BrandAssetsMenu } from "@/components/brand-assets-menu"
```

```tsx
<BrandAssetsMenu
  logomark={<OmPatilMark />}
  logomarkSVG="<svg>...</svg>"
  logotypeSVG="<svg>...</svg>"
  brandGuidelinesURL="https://heyompatil.me/components/brand-assets-menu"
  brandAssetsURL="/icon.svg"
>
  <Link />
</BrandAssetsMenu>
```
